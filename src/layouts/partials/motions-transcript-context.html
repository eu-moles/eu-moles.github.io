{{- $file := index . "file" -}} {{- $queries := index . "titles" -}} {{- if not (reflect.IsSlice $queries) -}} {{- $queries = slice $queries -}} {{- end -}} {{- $normalizedQueries := slice -}} {{- range $queries -}} {{- $query := lower (strings.TrimSpace .) -}} {{- $query = replaceRE "^[0-9]+[.[:space:]-]+" "" $query -}} {{- if ne $query "" -}} {{- $normalizedQueries = $normalizedQueries | append $query -}} {{- end -}} {{- end -}} {{- $turns := slice -}} {{- if and (gt (len $normalizedQueries) 0) (os.FileExists $file) -}} {{- $contexts := partials.IncludeCached "motions-transcript-contexts.html" $file $file -}} {{- range $contexts -}} {{- if eq (len $turns) 0 -}} {{- $contextHeading := .heading -}} {{- $contextTurns := .turns -}} {{- range $normalizedQueries -}} {{- if strings.Contains $contextHeading . -}} {{- $turns = $contextTurns -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "turns" $turns) -}}
