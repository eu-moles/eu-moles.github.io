{{- $file := . -}} {{- $languages := dict -}} {{- $english := "http://publications.europa.eu/resource/authority/language/ENG" -}} {{- if os.FileExists $file -}} {{- $data := transform.Unmarshal (os.ReadFile $file) -}} {{- range $data.data -}} {{- range .recorded_in_a_realization_of -}} {{- $sourceLanguages := .originalLanguage | default slice -}} {{- $languageCodes := slice -}} {{- $languageNames := slice -}} {{- $translationSource := "" -}} {{- range $uri := $sourceLanguages -}} {{- $language := partials.IncludeCached "transcript/transcript-language-code.html" $uri $uri -}} {{- if gt (len $language) 0 -}} {{- $languageCodes = $languageCodes | append (index $language "code") -}} {{- $languageNames = $languageNames | append (index $language "name") -}} {{- if and (ne $uri $english) (eq $translationSource "") -}} {{- $translationSource = lower (index $language "code") -}} {{- end -}} {{- end -}} {{- end -}} {{- if and .number (ne $translationSource "") -}} {{- $languages = merge $languages (dict .number (dict "code" (delimit $languageCodes " & ") "name" (delimit $languageNames " and ") "source" $translationSource )) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $languages -}}
