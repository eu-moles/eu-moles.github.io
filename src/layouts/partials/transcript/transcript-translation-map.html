{{- $file := . -}} {{- $translations := dict -}} {{- if os.FileExists $file -}} {{- $data := transform.Unmarshal (os.ReadFile $file) -}} {{- with index $data "translations" -}} {{- range $speechNumber, $translation := . -}} {{- if and $translation (ne (index $translation "englishText" | default "") "") -}} {{- $englishText := index $translation "englishText" -}} {{- /* Display cleanup for translations cached before the parser fix. */ -}} {{- $englishText = replaceRE `(?i)^[[:space:]]*,?[[:space:]]*on behalf of (the )?[^.\r\n]+\.[[:space:]]*[–—-][[:space:]]*` "" $englishText -}} {{- $translations = merge $translations (dict $speechNumber (merge $translation (dict "englishText" $englishText))) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $translations -}}
