{{- $uri := . -}} {{- $sourceCode := replaceRE "^.*/" "" $uri -}} {{- $file := printf "data/languages/%s.xml" $sourceCode -}} {{- $code := "" -}} {{- $name := "" -}} {{- $language := dict -}} {{- if os.FileExists $file -}} {{- $xml := os.ReadFile $file -}} {{- $codeMatches := findRE `[^<]+` $xml -}} {{- if gt (len $codeMatches) 0 -}} {{- $code = index $codeMatches 0 -}} {{- $code = replaceRE `^<[^>]+>` "" $code -}} {{- $code = replaceRE `]+>$` "" $code -}} {{- $code = upper $code -}} {{- end -}} {{- $nameMatches := findRE `[^<]+` $xml -}} {{- if gt (len $nameMatches) 0 -}} {{- $name = index $nameMatches 0 -}} {{- $name = replaceRE `^<[^>]+>` "" $name -}} {{- $name = replaceRE `]+>$` "" $name -}} {{- end -}} {{- end -}} {{- if ne $code "" -}} {{- $language = dict "code" $code "name" (default $code $name) -}} {{- end -}} {{- return $language -}}
